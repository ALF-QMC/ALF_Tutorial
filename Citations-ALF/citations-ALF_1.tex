\newcolumntype{L}[1]{%
  >{\raggedright\let\newline\\\arraybackslash\hspace{0pt}}p{#1}}
\newcolumntype{C}[1]{%
  >{\centering\let\newline\\\arraybackslash\hspace{0pt}}p{#1}}

  {\setlength{\LTpre}{0pt}%
   \setlength{\LTpost}{0pt}%
   \renewcommand*{\arraystretch}{1.2}%
   \begin{longtable}{%
                     @{}
                     L{\dimexpr0.25\textwidth-\tabcolsep\relax}
                     L{\dimexpr0.6\textwidth-2\tabcolsep\relax}
                     L{\dimexpr0.15\textwidth-\tabcolsep\relax}
                     @{}}
     \toprule
     \textbf{Reference} & \textbf{Title} & \textbf{ALF role}\\
     \midrule  \endhead}

   \end{longtable}}
